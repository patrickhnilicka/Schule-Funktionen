%Theorem
\newcounter{theo}[section]
\setcounter{theo}{0}
\renewcommand{\thetheo}{\arabic{section}.\arabic{theo}}

\newenvironment{theo}[2][]{%
  \refstepcounter{theo}%
  \ifstrempty{#1}{%
    \mdfsetup{%
      frametitle={%
          \tikz[baseline=(current bounding box.east),outer sep=0pt]
          \node[anchor=east,rectangle,fill=blue!50]{\strut Satz~\thetheo};}}%
  }{%
    \mdfsetup{%
      frametitle={%
          \tikz[baseline=(current bounding box.east),outer sep=0pt]
          \node[anchor=east,rectangle,fill=blue!50]{\strut Satz~\thetheo:~#1};}}%
  }%
  \mdfsetup{
    innertopmargin=10pt,
    linecolor=blue!50,
    linewidth=2pt,
    topline=true,
    frametitleaboveskip=\dimexpr-\ht\strutbox\relax,
    nobreak=true % kein Seitenumbruch direkt vor/nach dem Rahmen
  }%
  \begin{mdframed}\relax
    \begin{minipage}{\linewidth} % verhindert Seitenumbruch innerhalb
      \label{#2}%
      }{%
    \end{minipage}%
  \end{mdframed}%
}
